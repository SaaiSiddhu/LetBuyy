## 1. Database Configuration

- **Engine:** PostgreSQL 16 via Supabase
- **Naming:** `snake_case` for tables and columns
- **Primary key:** UUID via `gen_random_uuid()`
- **Timestamps:** `created_at TIMESTAMPTZ DEFAULT NOW()`, `updated_at TIMESTAMPTZ DEFAULT NOW()`
- **Trigger:** `update_updated_at()` — auto-manages `updated_at` column

## 3. Key Table Schemas (Detailed)

### `sites` (Platform Core)

```sql
CREATE TABLE sites (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  name VARCHAR(255) NOT NULL,
  subdomain VARCHAR(63) NOT NULL UNIQUE,    -- {subdomain}.letbuyy.store
  custom_domain VARCHAR(255),               -- custom domain (optional)
  site_type VARCHAR(20) NOT NULL CHECK (site_type IN ('store','portfolio','resume','blog','landing_page')),
  status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active','suspended','archived')),
  plan_id UUID REFERENCES plans(id),
  settings JSONB DEFAULT '{}',              -- store-specific configuration
  seo_title VARCHAR(60),
  seo_description VARCHAR(160),
  favicon_url TEXT,
  logo_url TEXT,
  published BOOLEAN DEFAULT false,
  trial_ends_at TIMESTAMPTZ,               -- NULL = no trial
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  deleted_at TIMESTAMPTZ                    -- PLANNED (soft delete)
);

-- Indexes
CREATE UNIQUE INDEX idx_sites_subdomain ON sites(subdomain) WHERE deleted_at IS NULL;
CREATE INDEX idx_sites_user_id ON sites(user_id);
CREATE INDEX idx_sites_site_type ON sites(site_type);
CREATE INDEX idx_sites_status ON sites(status);
```

### `products` (E-Commerce)

```sql
CREATE TABLE products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  site_id UUID NOT NULL REFERENCES sites(id) ON DELETE CASCADE,
  title VARCHAR(255) NOT NULL,
  slug VARCHAR(255) NOT NULL,
  description TEXT,                         -- Tiptap JSON
  short_description VARCHAR(500),
  price DECIMAL(10,2) NOT NULL,
  compare_at_price DECIMAL(10,2),           -- Original price (discounts)
  cost DECIMAL(10,2),                       -- Cost of goods (profit calc)
  sku VARCHAR(100),
  barcode VARCHAR(100),                     -- UPC/EAN/ISBN
  category_id UUID REFERENCES categories(id),
  status VARCHAR(20) DEFAULT 'draft' CHECK (status IN ('draft','active','archived')),
  weight_grams INTEGER,
  dimensions JSONB,                         -- { length, width, height } cm
  seo_title VARCHAR(60),
  seo_description VARCHAR(160),
  view_count INTEGER DEFAULT 0,
  sales_count INTEGER DEFAULT 0,
  track_inventory BOOLEAN DEFAULT true,
  allow_backorder BOOLEAN DEFAULT false,
  published_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  deleted_at TIMESTAMPTZ                    -- PLANNED (soft delete)
);

-- Indexes
CREATE UNIQUE INDEX idx_products_slug ON products(site_id, slug) WHERE deleted_at IS NULL;
CREATE INDEX idx_products_site_id ON products(site_id);
CREATE INDEX idx_products_category ON products(category_id);
CREATE INDEX idx_products_status ON products(status);
CREATE INDEX idx_products_search ON products USING GIN (to_tsvector('english', title || ' ' || COALESCE(description, '')));
```

### `orders` (E-Commerce)

```sql
CREATE TABLE orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  site_id UUID NOT NULL REFERENCES sites(id),
  order_number SERIAL,                      -- Auto-increment per store
  customer_id UUID REFERENCES store_customers(id),
  customer_email VARCHAR(255) NOT NULL,
  customer_name VARCHAR(255) NOT NULL,
  status VARCHAR(20) DEFAULT 'pending' CHECK (status IN ('pending','confirmed','processing','shipped','delivered','cancelled','refunded')),
  items JSONB NOT NULL,                     -- CURRENT: JSONB array → PLANNED: order_items table
  subtotal DECIMAL(10,2) NOT NULL,
  shipping_cost DECIMAL(10,2) DEFAULT 0,
  discount_amount DECIMAL(10,2) DEFAULT 0,
  tax_amount DECIMAL(10,2) DEFAULT 0,
  total DECIMAL(10,2) NOT NULL,
  currency VARCHAR(3) DEFAULT 'INR',
  shipping_address JSONB,
  billing_address JSONB,
  payment_method VARCHAR(20),
  payment_id VARCHAR(255),                  -- Gateway transaction ID
  payment_status VARCHAR(20) DEFAULT 'unpaid',
  shipped_at TIMESTAMPTZ,
  delivered_at TIMESTAMPTZ,
  notes TEXT,
  coupon_id UUID REFERENCES coupons(id),
  idempotency_key VARCHAR(255),
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  deleted_at TIMESTAMPTZ                    -- PLANNED (soft delete)
);

-- Indexes
CREATE INDEX idx_orders_site_id ON orders(site_id);
CREATE INDEX idx_orders_customer ON orders(customer_id);
CREATE INDEX idx_orders_status ON orders(status);
CREATE INDEX idx_orders_created ON orders(created_at DESC);
```

## 5. RLS Policies (Detailed)

```sql
-- Store owner can CRUD their own store's products
CREATE POLICY products_owner ON products
  FOR ALL USING (
    site_id IN (SELECT id FROM sites WHERE user_id = auth.uid())
  );

-- Published products readable by anyone (storefront)
CREATE POLICY products_public_read ON products
  FOR SELECT USING (
    status = 'active' AND deleted_at IS NULL
    AND site_id IN (SELECT id FROM sites WHERE published = true)
  );

-- Orders: owner read + write
CREATE POLICY orders_owner ON orders
  FOR ALL USING (
    site_id IN (SELECT id FROM sites WHERE user_id = auth.uid())
  );

-- Customers cannot see other customers' orders
CREATE POLICY orders_customer ON orders
  FOR SELECT USING (
    customer_id IN (SELECT id FROM store_customers WHERE user_id = auth.uid())
  );
```
